CREATE OR REPLACE FUNCTION total_spending(cid INT)
RETURNS NUMERIC AS $$
BEGIN
    RETURN (
        SELECT SUM(Total)
        FROM Invoice
        WHERE customer_id = cid
    );
END;
$$ LANGUAGE plpgsql;

SELECT total_spending(5);


CREATE OR REPLACE PROCEDURE new_manager(eid INT, nmid INT)
AS $$
BEGIN
    IF eid = nmid THEN
        RAISE EXCEPTION 'Employee cannot self report';
    END IF;

    IF NOT EXISTS (SELECT 1 FROM employee WHERE employee_id = nmid) THEN
        RAISE EXCEPTION 'Managerdoes not exist';
    END IF;

    IF EXISTS (SELECT 1 FROM employee WHERE employee_id = nmid AND reports_to = eid) 
    THEN
        RAISE EXCEPTION 'This creates a circular management relationship';
    END IF;

    UPDATE employee
    SET reports_to = nmid
    WHERE employee_id = eid;
END;
$$ LANGUAGE plpgsql;

CALL new_manager(5, 3);
